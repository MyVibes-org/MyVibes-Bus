# Rapid Bus PWA v2 - Implementation Plan

## Executive Summary

This document outlines the re-architecture of the Rapid Bus PWA to properly support real-time bus tracking with direction-aware notifications. The app uses official data.gov.my GTFS APIs as the single source of truth.

**Key Decisions Made:**
- `public/data/` in `.gitignore` - generated at build time
- Vercel prebuild script fetches GTFS and generates JSON
- Simple one-time notifications (like reminders)
- Browser Notifications (PWA-friendly)
- localStorage for alerts (no auth)

---

## Current Issues & Fixes

| Issue | Impact | Solution |
|-------|--------|----------|
| Default route `U4780` doesn't exist | App shows no data | Change to `U4000` (Bus 400) |
| `adm-zip` fails in serverless | Webhook crashes | Replace with `jszip` |
| Direction not detected | Wrong notifications | Use `direction_id` from GTFS |
| Single alert only | Limited utility | Multiple one-time alerts |
| Webhook publicly visible | Security concern | Remove UI, use prebuild only |
| 100+ JSON files in git | Repo bloat | Generate at build time |

### Vercel Build Flow
1. `npm install` (cached by Vercel)
2. `prebuild` → `npm run generate-gtfs` → generates `public/data/*`
3. `npm run build` → Next.js build (includes generated static files)
4. Deploy

**Caching Behavior:**
- Vercel caches `node_modules` between builds
- Generated `public/data/` files are rebuilt each deploy
- To force fresh GTFS: redeploy or set `VERCEL_FORCE_NO_BUILD_CACHE=1`

// app/api/buses/route.ts
export async function GET(request: NextRequest) {
  const categories = ['rapid-bus-kl', 'rapid-bus-mrtfeeder'];
  
  const allBuses: Bus[] = [];
  
  for (const category of categories) {
    const response = await fetch(
      `https://api.data.gov.my/gtfs-realtime/vehicle-position/prasarana?category=${category}`,
      { next: { revalidate: 15 } } // Cache for 15 seconds
    );
    
    const buffer = await response.arrayBuffer();
    const feed = GtfsRealtimeBindings.transit_realtime.FeedMessage.decode(
      new Uint8Array(buffer)
    );
    
    for (const entity of feed.entity) {
      if (!entity.vehicle) continue;
      
      const v = entity.vehicle;
      const tripId = v.trip?.tripId || '';
      const tripInfo = tripsCache[tripId]; // Loaded from trips.json
      
      allBuses.push({
        id: v.vehicle?.id || '',
        tripId,
        routeId: v.trip?.routeId || tripInfo?.routeId || '',
        routeShortName: getRouteShortName(v.trip?.routeId),
        directionId: tripInfo?.directionId || '0',
        headsign: tripInfo?.headsign || '',
        lat: v.position?.latitude || 0,
        lon: v.position?.longitude || 0,
        bearing: v.position?.bearing || 0,
        speed: v.position?.speed || 0,
        timestamp: Number(v.timestamp) || Date.now(),
      });
    }
  }
  
  return NextResponse.json({ 
    buses: allBuses,
    timestamp: Date.now()
  });
}

### 3. Alert Manager (localStorage)

```tsx
// lib/alertManager.ts
const STORAGE_KEY = 'rapidbus_alerts';

export function getAlerts(): Alert[] {
  const data = localStorage.getItem(STORAGE_KEY);
  return data ? JSON.parse(data) : [];
}

export function addAlert(alert: Omit<Alert, 'id' | 'createdAt' | 'status'>): Alert {
  const alerts = getAlerts();
  const newAlert: Alert = {
    ...alert,
    id: crypto.randomUUID(),
    createdAt: Date.now(),
    status: 'active',
  };
  alerts.push(newAlert);
  localStorage.setItem(STORAGE_KEY, JSON.stringify(alerts));
  return newAlert;
}

export function removeAlert(id: string): void {
  const alerts = getAlerts().filter(a => a.id !== id);
  localStorage.setItem(STORAGE_KEY, JSON.stringify(alerts));
}

export function markAlertTriggered(id: string): void {
  const alerts = getAlerts().map(a => 
    a.id === id ? { ...a, status: 'triggered' as const } : a
  );
  localStorage.setItem(STORAGE_KEY, JSON.stringify(alerts));
}

export function cleanupTriggeredAlerts(): void {
  // Remove alerts that were triggered more than 1 hour ago
  const oneHourAgo = Date.now() - 60 * 60 * 1000;
  const alerts = getAlerts().filter(a => 
    a.status !== 'triggered' || a.createdAt > oneHourAgo
  );
  localStorage.setItem(STORAGE_KEY, JSON.stringify(alerts));
}
```

### 4. Notification Checker Hook

```tsx
// hooks/useAlertChecker.ts
export function useAlertChecker(buses: Bus[], routeDetails: RouteDetail) {
  const alerts = useAlerts(); // Custom hook that syncs with localStorage
  
  useEffect(() => {
    const activeAlerts = alerts.filter(a => a.status === 'active');
    
    for (const alert of activeAlerts) {
      // Find buses matching this alert's route AND direction
      const matchingBuses = buses.filter(b => 
        b.routeId === alert.routeId && 
        b.directionId === alert.directionId
      );
      
      if (matchingBuses.length === 0) continue;
      
      // Get the stop and route shape for this direction
      const direction = routeDetails?.directions[alert.directionId];
      if (!direction) continue;
      
      // Calculate ETA for each matching bus
      for (const bus of matchingBuses) {
        const eta = estimateETA(
          { lat: bus.lat, lon: bus.lon },
          { lat: getStopLat(alert.stopId), lon: getStopLon(alert.stopId) },
          direction.shape,
          bus.speed * 3.6, // m/s to km/h
          direction.stops,
          alert.stopId
        );
        
        // Check if ETA matches the alert threshold
        if (eta !== null && eta <= alert.minutesBefore && eta > 0) {
          triggerNotification(alert, eta, bus);
          markAlertTriggered(alert.id);
          break; // One notification per alert
        }
      }
    }
  }, [buses, alerts, routeDetails]);
}

function triggerNotification(alert: Alert, eta: number, bus: Bus) {
  if (Notification.permission === 'granted') {
    new Notification(`Bus ${alert.routeShortName} Arriving!`, {
      body: `${Math.ceil(eta)} min to ${alert.stopName}\nHeading: ${alert.headsign}`,
      icon: '/icons/bus-icon.png',
      tag: alert.id, // Prevents duplicate notifications
    });
  }
}
```

## Direction-Aware ETA Calculation

### Enhanced Algorithm

```typescript
// lib/routeLogic.ts
export function estimateETA(
  busLocation: Point,
  stopLocation: Point,
  shape: [number, number][],
  speedKmH: number = 30,
  stopsSequence: StopOnRoute[],
  targetStopId: string
): number | null {
  
  // 1. Find bus position along route
  const busDistAlongRoute = getDistanceAlongRoute(busLocation, shape);
  
  // 2. Find target stop position along route
  const stopDistAlongRoute = getDistanceAlongRoute(stopLocation, shape);
  
  // 3. If bus is past the stop, return null (bus has passed)
  if (busDistAlongRoute >= stopDistAlongRoute) {
    return null; // Already passed
  }
  
  // 4. Calculate remaining distance
  const remainingDist = stopDistAlongRoute - busDistAlongRoute; // km
  
  // 5. Calculate ETA
  const effectiveSpeed = speedKmH > 0 ? speedKmH : 20; // Default 20 km/h
  const etaHours = remainingDist / effectiveSpeed;
  const etaMinutes = etaHours * 60;
  
  return etaMinutes;
}
```
